cmake_minimum_required(VERSION 3.20)
project(notes_fat_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE)
set(NFS "$ENV{ESP32_NFS_LIBRARY}")
set(LIBS "$ENV{ARDUINO_LIBRARIES}")
execute_process(COMMAND python3 "${ROOT}/tools/prepare_sdfat.py" "${LIBS}/SdFat" OUTPUT_VARIABLE NOTES_SDFAT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE SDFAT_STATUS)
if(NOT SDFAT_STATUS EQUAL 0)
 message(FATAL_ERROR "Pinned SdFat unavailable")
endif()
file(GLOB FAT "${NOTES_SDFAT}/src/FatLib/*.cpp" "${NOTES_SDFAT}/src/ExFatLib/*.cpp" "${NOTES_SDFAT}/src/FsLib/*.cpp" "${NOTES_SDFAT}/src/common/*.cpp")
add_executable(notes_fat_test main.cpp ${FAT} "${NFS}/src/Names.cpp" "${NFS}/src/Memory.cpp")
target_include_directories(notes_fat_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/shim" "${ROOT}/firmware/WishOS/src" "${ROOT}/hardware/agent-os" "${NFS}/src" "${NOTES_SDFAT}/src" "${LIBS}/ArduinoJson/src")
target_compile_definitions(notes_fat_test PRIVATE ENABLE_ARDUINO_FEATURES=1 ARDUINO=10819 ENABLE_ARDUINO_SERIAL=0 ENABLE_ARDUINO_STRING=0 SDFAT_FILE_TYPE=3 SPI_DRIVER_SELECT=3 USE_BLOCK_DEVICE_INTERFACE=1 USE_FCNTL_H=1 USE_UTF8_LONG_NAMES=1 ARDUINOJSON_ENABLE_ARDUINO_STREAM=0 ARDUINOJSON_ENABLE_ARDUINO_STRING=0 ARDUINOJSON_ENABLE_ARDUINO_PRINT=0 ARDUINOJSON_ENABLE_PROGMEM=0)
target_compile_options(notes_fat_test PRIVATE -UNDEBUG -U__ARM_FEATURE_UNALIGNED -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(notes_fat_test PRIVATE -fsanitize=address,undefined)
