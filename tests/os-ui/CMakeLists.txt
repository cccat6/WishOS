cmake_minimum_required(VERSION 3.20)
project(notes_ui_test LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(NOTES_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(DEFINED ENV{ARDUINO_LIBRARIES} AND NOT "$ENV{ARDUINO_LIBRARIES}" STREQUAL "")
  set(NOTES_ARDUINO_LIBS "$ENV{ARDUINO_LIBRARIES}" CACHE PATH "Arduino dependency root" FORCE)
elseif(NOT DEFINED NOTES_ARDUINO_LIBS)
  set(NOTES_ARDUINO_LIBS "$ENV{HOME}/Documents/Arduino/libraries" CACHE PATH "Arduino dependency root")
endif()
set(LV_BUILD_CONF_PATH "${NOTES_ROOT}/firmware/AgenticOS/src/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${NOTES_ARDUINO_LIBS}/lvgl" lvgl)
add_executable(notes_ui_test main.cpp "${NOTES_ROOT}/firmware/AgenticOS/src/Content.cpp" "${NOTES_ARDUINO_LIBS}/Adafruit_GFX_Library/Adafruit_GFX.cpp")
target_include_directories(notes_ui_test PRIVATE "${NOTES_ROOT}/firmware/AgenticOS/src" "${NOTES_ROOT}/tests/host" "${NOTES_ARDUINO_LIBS}/ArduinoJson/src" "${NOTES_ARDUINO_LIBS}/Adafruit_GFX_Library")
target_compile_definitions(notes_ui_test PRIVATE ARDUINO=10819 ARDUINOJSON_ENABLE_ARDUINO_STRING=0 ARDUINOJSON_ENABLE_ARDUINO_STREAM=0 ARDUINOJSON_ENABLE_ARDUINO_PRINT=0 ARDUINOJSON_ENABLE_PROGMEM=0)
target_link_libraries(notes_ui_test PRIVATE lvgl::lvgl)

target_compile_options(notes_ui_test PRIVATE -UNDEBUG)

target_compile_options(notes_ui_test PRIVATE -g -fsanitize=address,undefined)
target_link_options(notes_ui_test PRIVATE -fsanitize=address,undefined)
file(GLOB LUA_SOURCES "${NOTES_ROOT}/firmware/AgenticOS/src/lua/*.c")
add_library(notes_lua STATIC ${LUA_SOURCES})
target_link_libraries(notes_ui_test PRIVATE notes_lua)
add_executable(notes_suite_test suite_test.cpp "${NOTES_ROOT}/firmware/AgenticOS/src/Content.cpp" "${NOTES_ARDUINO_LIBS}/Adafruit_GFX_Library/Adafruit_GFX.cpp")
target_include_directories(notes_suite_test PRIVATE "${NOTES_ROOT}/firmware/AgenticOS/src" "${NOTES_ROOT}/tests/host" "${NOTES_ARDUINO_LIBS}/ArduinoJson/src" "${NOTES_ARDUINO_LIBS}/Adafruit_GFX_Library")
target_compile_definitions(notes_suite_test PRIVATE ARDUINO=10819 ARDUINOJSON_ENABLE_ARDUINO_STRING=0 ARDUINOJSON_ENABLE_ARDUINO_STREAM=0 ARDUINOJSON_ENABLE_ARDUINO_PRINT=0 ARDUINOJSON_ENABLE_PROGMEM=0)
target_compile_options(notes_suite_test PRIVATE -UNDEBUG -g -fsanitize=address,undefined)
target_link_options(notes_suite_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(notes_suite_test PRIVATE notes_lua lvgl::lvgl)
add_executable(notes_lua_test lua_test.cpp)
target_include_directories(notes_lua_test PRIVATE "${NOTES_ROOT}/firmware/AgenticOS/src" "${NOTES_ARDUINO_LIBS}/ArduinoJson/src")
target_compile_options(notes_lua_test PRIVATE -UNDEBUG -g -fsanitize=address,undefined)
target_link_options(notes_lua_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(notes_lua_test PRIVATE notes_lua)
add_executable(notes_memory_test memory_regions.cpp)
target_include_directories(notes_memory_test PRIVATE "${NOTES_ROOT}/firmware/AgenticOS/src" "${NOTES_ARDUINO_LIBS}/ArduinoJson/src")
target_compile_options(notes_memory_test PRIVATE -UNDEBUG -g -fsanitize=address,undefined)
target_link_options(notes_memory_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(notes_memory_test PRIVATE notes_lua)

add_executable(notes_app_test app_integration.cpp)
target_include_directories(notes_app_test PRIVATE "${NOTES_ROOT}/firmware/AgenticOS/src" "${NOTES_ARDUINO_LIBS}/ArduinoJson/src")
target_compile_options(notes_app_test PRIVATE -UNDEBUG -g -fsanitize=address,undefined)
target_link_options(notes_app_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(notes_app_test PRIVATE notes_lua)

add_executable(notes_failure_test failure_test.cpp)
target_include_directories(notes_failure_test PRIVATE "${NOTES_ROOT}/firmware/AgenticOS/src" "${NOTES_ARDUINO_LIBS}/ArduinoJson/src")
target_compile_options(notes_failure_test PRIVATE -UNDEBUG -g -fsanitize=address,undefined)

target_link_libraries(notes_failure_test PRIVATE notes_lua)
