# Compile the same portable application/renderer without any E6 header or source.
foreach(SURFACE IN ITEMS 128x64 320x240 239x321)
  string(REPLACE "x" ";" GEOMETRY "${SURFACE}")
  list(GET GEOMETRY 0 WIDTH)
  list(GET GEOMETRY 1 HEIGHT)
  set(TARGET notes_portable_ui_${SURFACE})
  add_executable(${TARGET} portable_display.cpp "${NOTES_ROOT}/firmware/WishOS/src/UiMemory.cpp" "${NOTES_ROOT}/firmware/WishOS/src/display/NativeText.cpp")
  target_include_directories(${TARGET} PRIVATE "${NOTES_ROOT}/firmware/WishOS/src" "${NOTES_ROOT}/tests/host" "${NOTES_ARDUINO_LIBS}/ArduinoJson/src")
  target_compile_definitions(${TARGET} PRIVATE TEST_UI_WIDTH=${WIDTH} TEST_UI_HEIGHT=${HEIGHT} ARDUINO=10819 ARDUINOJSON_ENABLE_ARDUINO_STRING=0 ARDUINOJSON_ENABLE_ARDUINO_STREAM=0 ARDUINOJSON_ENABLE_ARDUINO_PRINT=0 ARDUINOJSON_ENABLE_PROGMEM=0)
  target_compile_options(${TARGET} PRIVATE -UNDEBUG -g -fsanitize=address,undefined)
  target_link_options(${TARGET} PRIVATE -fsanitize=address,undefined)
  target_link_libraries(${TARGET} PRIVATE notes_lua lvgl::lvgl)
endforeach()
